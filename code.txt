/*
 * =====================================================
 *  Plant Growth Monitor — ESP32-CAM WITH NTP
 *
 *  BOARD  : AI Thinker ESP32-CAM
 *  SENSOR : DHT22 on IO13
 *
 *  SCHEDULE:
 *    Sensor log  → once every hour
 *    Photo       → on boot + daily at PHOTO_HOUR:PHOTO_MIN
 *
 *  SD CARD:
 *    /log.csv
 *    /photo_YYYY-MM-DD_HH-MM.jpg
 * =====================================================
 */

// ── Scheduled photo time ─────────────────────────────
const int PHOTO_HOUR = 12;
const int PHOTO_MIN  = 15;


// ── Libraries ─────────────────────────────────────────
#include "esp_camera.h"
#include "FS.h"
#include "SD_MMC.h"
#include "DHT.h"
#include "time.h"
#include "WiFi.h"
#include <math.h>


// ── WiFi ──────────────────────────────────────────────
#define WIFI_SSID    "WrongOne_Guest"
#define WIFI_PASS    "Nope5768"


// ── Timezone ──────────────────────────────────────────
// Pacific Time
#define GMT_OFFSET_SEC    (-7 * 3600)
#define DAYLIGHT_OFFSET   0
#define NTP_SERVER        "pool.ntp.org"


// ── DHT22 ─────────────────────────────────────────────
#define DHTPIN  13
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);


// ── LED ───────────────────────────────────────────────
#define LED_PIN 4


// ── Camera pins (AI Thinker ESP32-CAM) ────────────────
#define PWDN_GPIO_NUM     32
#define RESET_GPIO_NUM    -1
#define XCLK_GPIO_NUM      0
#define SIOD_GPIO_NUM     26
#define SIOC_GPIO_NUM     27

#define Y9_GPIO_NUM       35
#define Y8_GPIO_NUM       34
#define Y7_GPIO_NUM       39
#define Y6_GPIO_NUM       36
#define Y5_GPIO_NUM       21
#define Y4_GPIO_NUM       19
#define Y3_GPIO_NUM       18
#define Y2_GPIO_NUM        5

#define VSYNC_GPIO_NUM    25
#define HREF_GPIO_NUM     23
#define PCLK_GPIO_NUM     22


// ── Status variables ──────────────────────────────────
unsigned long photoCount = 0;

bool sdReady = false;
bool camReady = false;
bool ntpSynced = false;


// ── Scheduler tracking ────────────────────────────────
int lastLoggedHour = -1;
int lastPhotoDay   = -1;


// ── Dew point calculation ─────────────────────────────
float dewPoint(float tempC, float humidity) {

  float a = 17.271f;
  float b = 237.7f;

  float gamma =
    ((a * tempC) / (b + tempC)) +
    log(humidity / 100.0f);

  return (b * gamma) / (a - gamma);
}


// ── Get time ──────────────────────────────────────────
bool getTime(struct tm &timeinfo) {

  if (!getLocalTime(&timeinfo))
    return false;

  return true;
}


// ── Timestamp string ──────────────────────────────────
String timestamp() {

  struct tm timeinfo;

  if (!getTime(timeinfo))
    return "NO_TIME";

  char buf[24];

  strftime(buf,
           sizeof(buf),
           "%Y-%m-%d %H:%M:%S",
           &timeinfo);

  return String(buf);
}


// ── WiFi + NTP setup ──────────────────────────────────
void initWiFi() {

  Serial.printf("[WiFi] Connecting to %s", WIFI_SSID);

  WiFi.begin(WIFI_SSID, WIFI_PASS);

  int tries = 0;

  while (WiFi.status() != WL_CONNECTED && tries++ < 20) {

    delay(500);
    Serial.print(".");
  }


  if (WiFi.status() == WL_CONNECTED) {

    Serial.println(" Connected");

    configTime(
      GMT_OFFSET_SEC,
      DAYLIGHT_OFFSET,
      NTP_SERVER
    );


    struct tm timeinfo;

    int attempts = 0;

    while (!getLocalTime(&timeinfo) && attempts++ < 10) {

      Serial.println("[NTP] Waiting...");
      delay(1000);
    }


    if (getLocalTime(&timeinfo)) {

      char buf[32];

      strftime(
        buf,
        sizeof(buf),
        "%Y-%m-%d %H:%M:%S",
        &timeinfo
      );

      Serial.printf("[NTP] Synced: %s\n", buf);

      ntpSynced = true;


      digitalWrite(LED_PIN, HIGH);
      delay(300);
      digitalWrite(LED_PIN, LOW);

    } else {

      Serial.println("[NTP] FAILED");
    }

  } else {

    Serial.println("\n[WiFi] FAILED");

    for (int i = 0; i < 5; i++) {

      digitalWrite(LED_PIN, HIGH);
      delay(300);

      digitalWrite(LED_PIN, LOW);
      delay(300);
    }
  }
}

// ── Camera init ───────────────────────────────────────
bool initCamera() {

  camera_config_t config;

  config.ledc_channel = LEDC_CHANNEL_0;
  config.ledc_timer   = LEDC_TIMER_0;

  config.pin_d0 = Y2_GPIO_NUM;
  config.pin_d1 = Y3_GPIO_NUM;
  config.pin_d2 = Y4_GPIO_NUM;
  config.pin_d3 = Y5_GPIO_NUM;
  config.pin_d4 = Y6_GPIO_NUM;
  config.pin_d5 = Y7_GPIO_NUM;
  config.pin_d6 = Y8_GPIO_NUM;
  config.pin_d7 = Y9_GPIO_NUM;

  config.pin_xclk = XCLK_GPIO_NUM;
  config.pin_pclk = PCLK_GPIO_NUM;

  config.pin_vsync = VSYNC_GPIO_NUM;
  config.pin_href  = HREF_GPIO_NUM;

  config.pin_sccb_sda = SIOD_GPIO_NUM;
  config.pin_sccb_scl = SIOC_GPIO_NUM;

  config.pin_pwdn  = PWDN_GPIO_NUM;
  config.pin_reset = RESET_GPIO_NUM;

  config.xclk_freq_hz = 20000000;

  config.pixel_format = PIXFORMAT_JPEG;

  config.frame_size = FRAMESIZE_UXGA;
  config.jpeg_quality = 10;
  config.fb_count = 1;


  if (esp_camera_init(&config) != ESP_OK) {

    Serial.println("[CAM] Init FAILED");
    return false;
  }


  Serial.println("[CAM] Ready");
  return true;
}


// ── SD init ───────────────────────────────────────────
bool initSD() {

  if (!SD_MMC.begin("/sdcard", true)) {

    Serial.println("[SD] Mount FAILED");
    return false;
  }


  if (SD_MMC.cardType() == CARD_NONE) {

    Serial.println("[SD] No card");
    return false;
  }


  Serial.println("[SD] Ready");


  if (!SD_MMC.exists("/log.csv")) {

    File f = SD_MMC.open("/log.csv", FILE_WRITE);

    if (f) {

      f.println(
        "timestamp,temperature_c,humidity_pct,heat_index_c,dew_point_c"
      );

      f.close();
    }
  }

  return true;
}


// ── Find last photo number ────────────────────────────
unsigned long getLastPhotoCount() {

  unsigned long highest = 0;

  File root = SD_MMC.open("/");


  while (true) {

    File entry = root.openNextFile();

    if (!entry)
      break;


    String name = String(entry.name());


    if (name.startsWith("photo_")) {

      highest++;
    }


    entry.close();
  }


  root.close();

  return highest;
}


// ── Sensor logging ────────────────────────────────────
void logSensor() {

  float h = dht.readHumidity();
  float t = dht.readTemperature();


  if (isnan(h) || isnan(t)) {

    Serial.println("[DHT] FAILED");
    return;
  }


  float hi = dht.computeHeatIndex(t, h, false);

  float dp = dewPoint(t, h);


  Serial.printf(
    "[DHT] %.1f C  %.1f %%  HI %.1f  DP %.1f\n",
    t, h, hi, dp
  );


  if (!sdReady)
    return;


  File f = SD_MMC.open("/log.csv", FILE_APPEND);


  if (!f)
    return;


  f.printf(
    "%s,%.2f,%.2f,%.2f,%.2f\n",
    timestamp().c_str(),
    t,
    h,
    hi,
    dp
  );


  f.close();

  Serial.println("[CSV] Logged");
}


// ── Photo capture ─────────────────────────────────────
void takePhoto() {

  if (!camReady || !sdReady)
    return;


  digitalWrite(LED_PIN, HIGH);
  delay(150);


  camera_fb_t *fb = esp_camera_fb_get();


  digitalWrite(LED_PIN, LOW);


  if (!fb) {

    Serial.println("[CAM] Capture FAILED");
    return;
  }


  char path[64];

  struct tm timeinfo;


  if (getTime(timeinfo)) {

    strftime(
      path,
      sizeof(path),
      "/photo_%Y-%m-%d_%H-%M.jpg",
      &timeinfo
    );

  } else {

    snprintf(
      path,
      sizeof(path),
      "/photo_%lu.jpg",
      photoCount++
    );
  }


  File f = SD_MMC.open(path, FILE_WRITE);


  if (f) {

    f.write(fb->buf, fb->len);
    f.close();

    Serial.printf(
      "[CAM] Saved %s\n",
      path
    );

  } else {

    Serial.println("[CAM] Save FAILED");
  }


  esp_camera_fb_return(fb);
}


// ── Helper: seconds until next HH:MM ──────────────────
long secondsUntil(int targetHour, int targetMin) {

  struct tm timeinfo;


  if (!getTime(timeinfo))
    return 60;


  int nowSec =
    timeinfo.tm_hour * 3600 +
    timeinfo.tm_min  * 60 +
    timeinfo.tm_sec;


  int targetSec =
    targetHour * 3600 +
    targetMin  * 60;


  long diff = targetSec - nowSec;


  if (diff <= 0)
    diff += 86400;


  return diff;
}


// ── Setup ─────────────────────────────────────────────
void setup() {

  Serial.begin(115200);

  delay(2000);


  Serial.println("\n=== PLANT MONITOR START ===");


  pinMode(LED_PIN, OUTPUT);

  digitalWrite(LED_PIN, LOW);


  dht.begin();


  delay(2000);


  initWiFi();


  camReady = initCamera();

  sdReady  = initSD();


  if (sdReady)
    photoCount = getLastPhotoCount();



  // Boot actions
  Serial.println("\n--- Boot reading ---");

  logSensor();

  takePhoto();



  // Prevent duplicate hourly log
  struct tm timeinfo;

  if (getTime(timeinfo)) {

    lastLoggedHour = timeinfo.tm_hour;
  }


  Serial.println("[OK] Running");
}



// ── Main loop ─────────────────────────────────────────
void loop() {


  struct tm timeinfo;


  if (!getTime(timeinfo)) {

    delay(60000);
    return;
  }


  int currentHour = timeinfo.tm_hour;
  int currentMin  = timeinfo.tm_min;
  int currentDay  = timeinfo.tm_mday;



  // Hourly sensor log
  if (currentHour != lastLoggedHour) {

    lastLoggedHour = currentHour;

    Serial.printf(
      "\n--- %s ---\n",
      timestamp().c_str()
    );

    logSensor();
  }



  // Scheduled daily photo
  if (currentDay != lastPhotoDay &&
      (currentHour > PHOTO_HOUR ||
      (currentHour == PHOTO_HOUR &&
       currentMin >= PHOTO_MIN))) {


    lastPhotoDay = currentDay;


    Serial.printf(
      "\n--- Scheduled photo %s ---\n",
      timestamp().c_str()
    );


    takePhoto();
  }



  // Decide next wake time

  long secsToLog =
    secondsUntil((currentHour + 1) % 24, 0);


  long secsToPhoto;


  if (currentDay == lastPhotoDay) {

    secsToPhoto =
      secondsUntil(PHOTO_HOUR, PHOTO_MIN);

  }

  else if (currentHour < PHOTO_HOUR ||
          (currentHour == PHOTO_HOUR &&
           currentMin < PHOTO_MIN)) {


    secsToPhoto =
      secondsUntil(PHOTO_HOUR, PHOTO_MIN);

  }

  else {

    secsToPhoto = 5;
  }



  long sleepSecs =
    min(secsToLog, secsToPhoto) - 5;


  if (sleepSecs < 1)
    sleepSecs = 1;



  Serial.printf(
    "[TIME] %02d:%02d:%02d | Log %ld sec | Photo %ld sec\n",
    currentHour,
    currentMin,
    timeinfo.tm_sec,
    secsToLog,
    secsToPhoto
  );


  Serial.printf(
    "[SLEEP] %ld min %ld sec\n",
    sleepSecs / 60,
    sleepSecs % 60
  );


  delay(sleepSecs * 1000UL);
}
