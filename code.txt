/*
 * ============================================================
 * ESP32-CAM PLANT MONITOR — INTEGRATED DIAGNOSTIC BUILD
 *
 * Board: AI Thinker ESP32-CAM (ESP32-CAM-MB)
 * Camera: VGA 640x480 JPEG, PSRAM, one frame buffer
 * SD: Built-in microSD, one-bit mode
 * DHT22: GPIO3 (UART RX; disconnect sensor if upload issues)
 * L298N: IN1 GPIO12, IN2 GPIO13 (GPIO 4 left free to avoid flash LED)
 *
 * Files:
 *   /log.csv
 *   /errors.txt
 *   /photo_YYYY-MM-DD_HH-MM.jpg
 *
 * Schedule:
 *   Every hour at :00 — sensor logging
 *   Daily at 12:15    — photo
 *   Every hour at :30 — dip cycle
 *
 * Actuator:
 *   Extend 20 seconds
 *   Hold 30 seconds
 *   Retract 20 seconds
 * ============================================================
 */

#include "esp_camera.h"
#include "FS.h"
#include "SD_MMC.h"
#include "DHT.h"
#include "WiFi.h"
#include "time.h"
#include <math.h>

// ---------------- Wi-Fi ----------------
#define WIFI_SSID    "WrongOne_Guest"
#define WIFI_PASS    "Nope5768"

// ---------------- Time -----------------
#define NTP_SERVER "pool.ntp.org"
#define TZ_INFO "PST8PDT,M3.2.0/2,M11.1.0/2"

// ---------------- DHT22 ----------------
#define DHT_PIN 3
#define DHT_TYPE DHT22
DHT dht(DHT_PIN, DHT_TYPE);

// ---------------- L298N ----------------
#define MOTOR_IN1 12  // Safe pin, avoids onboard flash LED on GPIO 4
#define MOTOR_IN2 13

#define EXTEND_MS  20000UL
#define HOLD_MS    30000UL
#define RETRACT_MS 20000UL

// ---------------- Camera pins -----------
#define PWDN_GPIO_NUM     32
#define RESET_GPIO_NUM    -1
#define XCLK_GPIO_NUM      0
#define SIOD_GPIO_NUM     26
#define SIOC_GPIO_NUM     27
#define Y9_GPIO_NUM       35
#define Y8_GPIO_NUM       34
#define Y7_GPIO_NUM       39
#define Y6_GPIO_NUM       36
#define Y5_GPIO_NUM       21
#define Y4_GPIO_NUM       19
#define Y3_GPIO_NUM       18
#define Y2_GPIO_NUM        5
#define VSYNC_GPIO_NUM    25
#define HREF_GPIO_NUM     23
#define PCLK_GPIO_NUM     22

// ---------------- State ----------------
bool sdReady = false;
bool cameraReady = false;
bool wifiReady = false;
bool timeReady = false;

int lastLoggedHour = -1;
int lastLoggedDay = -1;
int lastPhotoDay = -1;
int lastDipHour = -1;
int lastDipDay = -1;

// ---------------- Watchdog Helper ------
void delayWithWDT(unsigned long ms) {
  unsigned long start = millis();
  while (millis() - start < ms) {
    delay(100); // Feeds the task watchdog timer
  }
}

// ---------------- Clock ----------------
bool readClock(struct tm &t) {
  return getLocalTime(&t, 1000);
}

String timestamp() {
  struct tm t;
  if (!readClock(t)) return "NO_TIME";

  char buf[24];
  strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &t);
  return String(buf);
}

// ---------------- Error logging --------
void logError(const String &message) {
  String ts = timestamp();
  Serial.printf("[ERROR] %s | %s\n", ts.c_str(), message.c_str());

  if (!sdReady) {
    Serial.println("[ERROR] SD unavailable; error not saved");
    return;
  }

  File f = SD_MMC.open("/errors.txt", FILE_APPEND);
  if (!f) {
    Serial.println("[ERROR] Could not open /errors.txt");
    return;
  }

  f.printf("%s, %s\n", ts.c_str(), message.c_str());
  f.close();
}

// ---------------- Motor -----------------
void stopMotor() {
  digitalWrite(MOTOR_IN1, LOW);
  digitalWrite(MOTOR_IN2, LOW);
}

void extendActuator() {
  Serial.println("[MOTOR] Extend START");
  digitalWrite(MOTOR_IN1, LOW);
  digitalWrite(MOTOR_IN2, HIGH);
  delayWithWDT(EXTEND_MS);
  stopMotor();
  Serial.println("[MOTOR] Extend COMPLETE");
}

void retractActuator() {
  Serial.println("[MOTOR] Retract START");
  digitalWrite(MOTOR_IN1, HIGH);
  digitalWrite(MOTOR_IN2, LOW);
  delayWithWDT(RETRACT_MS);
  stopMotor();
  Serial.println("[MOTOR] Retract COMPLETE");
}

void dipCycle() {
  Serial.printf("[DIP] START at %s\n", timestamp().c_str());
  extendActuator();
  Serial.println("[DIP] Holding for 30 seconds");
  delayWithWDT(HOLD_MS);
  retractActuator();
  Serial.printf("[DIP] COMPLETE at %s\n", timestamp().c_str());
}

// ---------------- Dew point -------------
float dewPoint(float tempC, float humidity) {
  if (humidity <= 0.0f || humidity > 100.0f) {
    return NAN;
  }
  const float a = 17.271f;
  const float b = 237.7f;
  float gamma = (a * tempC) / (b + tempC) + logf(humidity / 100.0f);
  return (b * gamma) / (a - gamma);
}

// ---------------- Camera ----------------
bool initCamera() {
  Serial.println("[CAM] Initializing camera");
  if (!psramFound()) {
    Serial.println("[CAM] PSRAM missing");
    return false;
  }

  camera_config_t c = {};
  c.ledc_channel = LEDC_CHANNEL_0;
  c.ledc_timer = LEDC_TIMER_0;
  c.pin_d0 = Y2_GPIO_NUM;
  c.pin_d1 = Y3_GPIO_NUM;
  c.pin_d2 = Y4_GPIO_NUM;
  c.pin_d3 = Y5_GPIO_NUM;
  c.pin_d4 = Y6_GPIO_NUM;
  c.pin_d5 = Y7_GPIO_NUM;
  c.pin_d6 = Y8_GPIO_NUM;
  c.pin_d7 = Y9_GPIO_NUM;
  c.pin_xclk = XCLK_GPIO_NUM;
  c.pin_pclk = PCLK_GPIO_NUM;
  c.pin_vsync = VSYNC_GPIO_NUM;
  c.pin_href = HREF_GPIO_NUM;
  c.pin_sccb_sda = SIOD_GPIO_NUM;
  c.pin_sccb_scl = SIOC_GPIO_NUM;
  c.pin_pwdn = PWDN_GPIO_NUM;
  c.pin_reset = RESET_GPIO_NUM;
  c.xclk_freq_hz = 10000000;
  c.pixel_format = PIXFORMAT_JPEG;
  c.frame_size = FRAMESIZE_VGA;
  c.jpeg_quality = 15;
  c.fb_count = 1;
  c.fb_location = CAMERA_FB_IN_PSRAM;
  c.grab_mode = CAMERA_GRAB_WHEN_EMPTY;

  esp_err_t err = esp_camera_init(&c);
  if (err != ESP_OK) {
    Serial.printf("[CAM] INIT FAILED: 0x%x\n", err);
    return false;
  }

  Serial.println("[CAM] VGA initialization successful");
  return true;
}

// ---------------- SD card ---------------
bool initSD() {
  Serial.println("[SD] Mounting built-in card, one-bit mode");
  if (!SD_MMC.begin("/sdcard", true)) {
    Serial.println("[SD] MOUNT FAILED");
    return false;
  }

  if (SD_MMC.cardType() == CARD_NONE) {
    Serial.println("[SD] NO CARD");
    SD_MMC.end();
    return false;
  }

  if (!SD_MMC.exists("/log.csv")) {
    File f = SD_MMC.open("/log.csv", FILE_WRITE);
    if (f) {
      f.println("timestamp,temperature_c,humidity_pct,heat_index_c,dew_point_c");
      f.close();
    }
  }

  if (!SD_MMC.exists("/errors.txt")) {
    File f = SD_MMC.open("/errors.txt", FILE_WRITE);
    if (f) {
      f.println("Plant monitor error log");
      f.close();
    }
  }

  Serial.println("[SD] Mount and file checks successful");
  return true;
}

// ---------------- Wi-Fi and NTP ---------
void initWiFi() {
  Serial.printf("[WiFi] Connecting to %s\n", WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);

  int tries = 0;
  while (WiFi.status() != WL_CONNECTED && tries < 30) {
    delay(500);
    Serial.print(".");
    ++tries;
  }
  Serial.println();

  if (WiFi.status() != WL_CONNECTED) {
    logError("WIFI CONNECTION FAILED");
    return;
  }

  wifiReady = true;
  Serial.printf("[WiFi] Connected: %s\n", WiFi.localIP().toString().c_str());

  configTzTime(TZ_INFO, NTP_SERVER);
  Serial.println("[NTP] Waiting for synchronization");

  struct tm t;
  for (int i = 0; i < 20; ++i) {
    if (readClock(t)) {
      timeReady = true;
      Serial.printf("[NTP] Synced: %s\n", timestamp().c_str());
      return;
    }
    delay(1000);
  }

  logError("NTP SYNC FAILED");
}

// ---------------- Photo ----------------
bool takePhoto() {
  if (!cameraReady || !sdReady) return false;

  camera_fb_t *fb = nullptr;
  for (int attempt = 1; attempt <= 3; ++attempt) {
    fb = esp_camera_fb_get();
    if (fb) break;
    delay(300);
  }

  if (!fb) {
    logError("CAMERA CAPTURE FAILED");
    return false;
  }

  char path[48];
  struct tm t;
  if (readClock(t)) {
    strftime(path, sizeof(path), "/photo_%Y-%m-%d_%H-%M.jpg", &t);
  } else {
    snprintf(path, sizeof(path), "/photo_notime.jpg");
  }

  bool saved = false;
  File f = SD_MMC.open(path, FILE_WRITE);
  if (f) {
    size_t written = f.write(fb->buf, fb->len);
    f.close();
    if (written == fb->len) {
      saved = true;
      Serial.printf("[CAM] PHOTO SAVED: %s\n", path);
    }
  }

  esp_camera_fb_return(fb);
  return saved;
}

// ---------------- Sensor logging --------
void logSensor() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();

  if (isnan(h) || isnan(t)) {
    logError("DHT22 READ FAILED");
    return;
  }

  float hi = dht.computeHeatIndex(t, h, false);
  float dp = dewPoint(t, h);

  if (!sdReady) return;

  File f = SD_MMC.open("/log.csv", FILE_APPEND);
  if (f) {
    f.printf("%s,%.2f,%.2f,%.2f,%.2f\n", timestamp().c_str(), t, h, hi, dp);
    f.close();
    Serial.println("[CSV] Sensor record saved");
  }
}

// ---------------- Setup -----------------
void setup() {
  Serial.begin(115200);
  delay(2000);

  Serial.println("\n========================================");
  Serial.println("PLANT MONITOR — REBUILT BUILD");
  Serial.println("========================================");

  pinMode(MOTOR_IN1, OUTPUT);
  pinMode(MOTOR_IN2, OUTPUT);
  stopMotor();

  dht.begin();
  delay(2000);

  Serial.println("[BOOT] Step 1: Initial actuator retraction");
  retractActuator();

  Serial.println("[BOOT] Step 2: Camera initialization");
  cameraReady = initCamera();

  Serial.println("[BOOT] Step 3: SD initialization");
  sdReady = initSD();

  Serial.println("[BOOT] Step 4: Wi-Fi and NTP");
  initWiFi();

  Serial.println("[BOOT] Step 5: Boot photo");
  if (cameraReady && sdReady) takePhoto();

  Serial.println("[BOOT] Step 6: Initial dip cycle");
  dipCycle();

  struct tm t;
  if (readClock(t)) {
    lastLoggedHour = t.tm_hour;
    lastLoggedDay = t.tm_yday;
    lastPhotoDay = t.tm_yday;

    int minsToNextDip = 30 - (t.tm_min % 30);
    if (minsToNextDip <= 5) {
      lastDipHour = t.tm_hour;
      lastDipDay = t.tm_yday;
      Serial.println("[BOOT] Skipping dip; too close to :30");
    }
  }

  Serial.println("[BOOT] Startup complete");
  Serial.println("========================================");
}

// ---------------- Main loop -------------
void loop() {
  struct tm t;
  if (!readClock(t)) {
    delay(1000);
    return;
  }

  timeReady = true;
  int hourNow = t.tm_hour;
  int minuteNow = t.tm_min;
  int dayNow = t.tm_yday;

  if (minuteNow == 0 && (hourNow != lastLoggedHour || dayNow != lastLoggedDay)) {
    lastLoggedHour = hourNow;
    lastLoggedDay = dayNow;
    logSensor();
  }

  if (hourNow == 12 && minuteNow == 15 && dayNow != lastPhotoDay) {
    lastPhotoDay = dayNow;
    takePhoto();
  }

  if (minuteNow == 30 && (hourNow != lastDipHour || dayNow != lastDipDay)) {
    lastDipHour = hourNow;
    lastDipDay = dayNow;
    dipCycle();
  }

  delay(500);
}
